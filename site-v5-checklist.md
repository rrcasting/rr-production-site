## 6. 言語切替とスティッキー CTA の動き

- 言語切替 `#lang-bar`: `.lang-btn[data-code=en|es|ja]` を押すと `setLang()`。全 `[data-lang]` に `.active` を付け外しし、CSS `[data-lang]{display:none}` / `.active{display:block}` / `span.active{display:inline}` で表示を切り替える。`<html lang>` も変える。選んだ言語は `localStorage rr_lang` に保存。初回は保存値 → 無ければブラウザ言語（es→ES、ja→JA、それ以外 EN）。
- ナビ: 768px 以下で `nav` を隠し `.hamburger`（`onclick="toggleMenu()"`）→ `#mobile-nav` に `.open`。
- スティッキー CTA `#sticky-cta`（`register.html` へのリンク）: 初期 `style="display:none;"`。読み込み時に画面幅 768px 以下なら、`#hero` と `#apply` を IntersectionObserver で見て、どちらかが見えている間は隠し、見えていなければ表示。
- FAQ: `.faq-q` の `onclick="toggleFaq(this)"`。開くと他は閉じる。`role=button` / `tabindex=0` / `aria-expanded` / Enter・Space で開閉。
- スクロール表示: `.reveal` を IntersectionObserver で `.visible` にする。

## 7. <script> の中身（旧）

```js
  /* ── Language switcher (persisted in localStorage) ── */
  let currentLang = 'en';

  function setLang(lang) {
    if (!['en', 'es', 'ja'].includes(lang)) lang = 'en';
    currentLang = lang;
    document.querySelectorAll('[data-lang]').forEach(el => {
      el.classList.toggle('active', el.dataset.lang === lang);
    });
    document.querySelectorAll('.lang-btn').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.code === lang);
    });
    document.documentElement.lang = lang === 'ja' ? 'ja' : lang === 'es' ? 'es' : 'en';
    try { localStorage.setItem('rr_lang', lang); } catch (e) {}
  }

  /* Init: saved language first, otherwise browser language (es→ES, ja→JA, else EN) */
  (function () {
    let saved = null;
    try { saved = localStorage.getItem('rr_lang'); } catch (e) {}
    if (saved && ['en', 'es', 'ja'].includes(saved)) {
      setLang(saved);
    } else {
      const nav = (navigator.language || navigator.userLanguage || 'en').slice(0, 2).toLowerCase();
      setLang(nav === 'es' ? 'es' : nav === 'ja' ? 'ja' : 'en');
    }
  })();

  /* ── Bind lang buttons via addEventListener for reliable Android tap ── */
  document.querySelectorAll('.lang-btn').forEach(btn => {
    btn.addEventListener('click', function() {
      const lang = (this.dataset.code || 'en');
      setLang(lang);
    });
  });

  /* ── Mobile nav ── */
  function toggleMenu() {
    document.getElementById('mobile-nav').classList.toggle('open');
  }

  /* ── FAQ accordion (with aria-expanded) ── */
  function toggleFaq(el) {
    const item = el.closest('.faq-item');
    const isOpen = item.classList.contains('open');
    document.querySelectorAll('.faq-item.open').forEach(i => {
      i.classList.remove('open');
      const q = i.querySelector('.faq-q');
      if (q) q.setAttribute('aria-expanded', 'false');
    });
    if (!isOpen) {
      item.classList.add('open');
      el.setAttribute('aria-expanded', 'true');
    }
  }

  /* FAQ accessibility: role, tabindex, aria-expanded, keyboard */
  document.querySelectorAll('.faq-q').forEach(q => {
    q.setAttribute('role', 'button');
    q.setAttribute('tabindex', '0');
    q.setAttribute('aria-expanded', 'false');
    q.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); toggleFaq(q); }
    });
  });

  /* ── Copy application template ── */


  /* ── Scroll reveal ── */
  const revealObserver = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if (e.isIntersecting) {
        e.target.classList.add('visible');
        revealObserver.unobserve(e.target);
      }
    });
  }, { threshold: 0.1 });
  document.querySelectorAll('.reveal').forEach(el => revealObserver.observe(el));

  /* ── Sticky CTA: show on mobile after hero scrolls out ── */
  if (window.innerWidth <= 768) {
    const stickyCta = document.getElementById('sticky-cta');
    const heroEl = document.getElementById('hero');
    const applyEl = document.getElementById('apply');

    const stickyObs = new IntersectionObserver((entries) => {
      entries.forEach(e => {
        if (e.target === heroEl || e.target === applyEl) {
          stickyCta.style.display = e.isIntersecting ? 'none' : 'block';
        }
      });
    }, { threshold: 0.1 });

    stickyObs.observe(heroEl);
    stickyObs.observe(applyEl);
  }
```
